\section{The effective theory prior to $\GF$ normalisation}
\begin{equation}
  \Lag_{\text{no-}\GF} =  \widehat{C}_{\mu}Q_\mu +V_{ud}\widehat{C}_{q} Q_q  + \rm{h.c.}
\end{equation}
The Wilson coefficients admit an expansion in terms the electromagnetic coupling
in the five-flavour theory, $\aef$,
\begin{align}
  \widehat{C}_X = 4\pi\aef\left(
                        \widehat{C}^\LO_{X} 
    + \frac{\aef}{4\pi} \widehat{C}^\NLO_{X} 
    + \bigg(\frac{\aef}{4\pi}\bigg)^2 \widehat{C}^\NNLO_{X} 
  + {\cal O}\big(\aef^3\big)\right)
  \quad\text{with}\quad X=\{\mu,q\}\,.
\end{align}
The value of the leading-order (LO) Wilson coefficients are
\begin{align}
  \widehat{C}^\LO_{X} = - \frac{1}{2\sw^2 \MW^2}
  \quad\text{with}\quad X=\{\mu,q\}\,.
  \label{<+label+>}
\end{align}


\section{The effective theory normalised to $\GF$}
Next we use $\GF$ as input and thus normalise the effective Lagrangian in terms of
the Wilson coefficient of the leptonic operator, i.e., we identify its 
Wilson coefficient with $\GF$,
\begin{equation}
  \Lag_{\GF} = -\frac{\GF}{\sqrt{2}} Q_\mu -\frac{\GF}{\sqrt{2}} V_{ud} C_{q} Q_q +\rm{h.c.}
\end{equation}
As before, $C_{q}$ admits an expansion in terms of $\aef$:
\begin{equation*}
  C_q = C_q^{\rm{LO}} + \frac{\aef}{4\pi} C_q^{\rm{NLO}} + \left(\frac{\aef}{4\pi}\right)^2 C_q^{\rm{NNLO}} + \dots
\end{equation*}
Comparing the two effective theories 
in perturbation theory in $\aef$ we readily find the relations
\begin{align}
   C_{q}^{\rm{LO}}   = 1\,,&
  &C_{q}^{\rm{NLO}}  = \frac{\widehat{C}_q^{\rm{NLO}} - \widehat{C}_\mu^{\rm{NLO}}}{\widehat{C}_\mu^{\rm{LO}}}\,,& 
  &C_{q}^{\rm{NNLO}} = \frac{
                             \widehat{C}_\mu^{\rm{LO}}\widehat{C}_q^{\rm{NNLO}} 
                            -\widehat{C}_\mu^{\rm{LO}}\widehat{C}_\mu^{\rm{NNLO}} 
                            -\widehat{C}_\mu^{\rm{NLO}}\widehat{C}_q^{\rm{NLO}} 
                            +(\widehat{C}_\mu^{\rm{NLO}})^2 
                          }{(\widehat{C}_\mu^{\rm{LO}})^2}\,,& 
  \label{<+label+>}
\end{align}
where we already used the relation $\widehat{C}_{\mu}^{\rm{LO}}=\widehat{C}_{q}^{\rm{LO}}$.


The above Wilson coefficient is scheme and scale dependent.
As this scale dependence only cancels after folding the result with the evolution kernel, we will define a regularisation independent Wilson coefficient.
This will result in manifest scheme and residual scale independent coefficient that has two obvious benefits:
It directly matches to the Fermi constant extracted from muon decay, and its scale dependence and perturbative converegence can be assessed independent form contributions at different scales.
Extending the definition of [GMJ] by including the renormalisation of the Fermi operator and three-loop QED running effects, we write
\begin{equation}
  \label{eq:chat-definition}
  \widehat{C}(\mu_h)= u_{5}^{-1}(\mu_h) J_{q,5}^{-1}J_{\mu,5}(\mu_h) C_q(\mu_h)\,,
\end{equation}
where the 
\begin{equation}
  \label{eq:chat-definition}
  J_{q,f}^{-1}J_{\mu,f}(\mu_h) = 
        1 
        + \frac{\aeF(\mu_h)}{4\pi}          \left[ J_{\mu,f}^{(1)}-J_{q,f}^{(1)} \right] 
        + \left(\frac{\aeF(\mu_h)}{4\pi}\right)^2 \left[ J_{\mu,f}^{(2)}-J_{q,f}^{(2)} - J_{\mu,f}^{(1)} J_{q,f}^{(1)} + \left( J_{q,f}^{(1)} \right)^2 \right]
  \,,
\end{equation}
with
\begin{equation}
  \label{eq:u-lo}
  u_f(\mu_h) =
  \left( \frac{\aeF(\mu_h)}{\aeF(M_Z^{\rm OS})} \right)^{-\frac{3}{2\overline{Q^2}}}
%  \left( \frac{\alpha^{(f)}_s(\mu)}{\alpha_s(M_Z)} \right)^{-\frac{\alpha}{4\pi}\frac{6}{33 - 2 n_f}}
\end{equation}
\begin{equation}
  \label{eq:j1}
  J_{x,f}^{(1)} = \frac{\gamma_x^{(1)}}{2 \beta_0}-\frac{\beta_1 \gamma_x^{(0)}}{2 \beta_0^2}
\end{equation}
and 
\begin{equation}
  \label{eq:j2}
  J_{x,f}^{(2)} = -\frac{\beta_2 \gamma_x^{(0)}}{4 \beta_0^2}+\frac{\gamma_x^{(2)}}{4
   \beta_0}-\frac{\beta_1 J_{x,1}^{(1)}}{2 \beta_0}+\frac{\left(J_{x,f}^{(1)} \right)^2}{2}
\end{equation}

Directly evaluating the NLO contribution leads to the very compact result
\begin{equation}
  C^{\rm NLO}_q = -5 +\frac{a^1_\mu}{4}+\frac{a^1_q}{12} - 2\log\left( \frac{\mu^2}{\MZ^2} \right)\,.
\end{equation}
